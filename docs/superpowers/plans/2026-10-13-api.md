# API Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Expose the ingested catalog to the front end: regions, an event list with preferred mechanisms, event detail with every product, ingest status and run history, and the three geo files, all as JSON with short cache headers and one error shape.

**Architecture:** DTO types live in `packages/shared/src/api.ts`, derived from the domain types with a `Json<T>` mapped type so dates become ISO strings. The server parses queries with zod (`src/api/schemas.ts`), maps domain objects to DTOs (`src/api/serialize.ts`), and registers one Fastify plugin per route under `src/routes`. An app-level error handler turns `HttpError` into `{ error, issues? }` and everything else into a logged 500. Repositories gain the few reads the routes need.

**Tech Stack:** No new dependencies. Fastify 5, zod 4, Drizzle, Vitest, Testcontainers, all already installed.

import type { BBox, StoredEvent, StoredMechanism } from "./index.js";

/** The wire form of a domain object: Date becomes an ISO 8601 UTC string. */
export type Json<T> = {
  [K in keyof T]: T[K] extends Date ? string : T[K] extends Date | null ? string | null : T[K];
};

export type ApiMechanism = Omit<Json<StoredMechanism>, "raw">;

/** A listed event. `mechanisms` holds the preferred product per kind. */
export interface ApiEvent extends Omit<
  Json<StoredEvent>,
  "raw" | "firstSeen" | "lastSeen" | "detailCheckedUpdated"
> {
  mechanisms: ApiMechanism[];
}

/** One event with every mechanism product and the raw USGS feed properties. */
export interface ApiEventDetail extends Omit<Json<StoredEvent>, "raw"> {
  mechanisms: ApiMechanism[];
  usgs: Record<string, unknown>;
}

/** The filters applied to a list request, after defaults. */
export interface ApiEventQuery {
  since: string;
  until: string | null;
  minMag: number | null;
  bbox: BBox | null;
  limit: number;
}

export interface ApiEventList {
  events: ApiEvent[];
  count: number;
  /** True when the result filled `limit`; narrow the window or raise the limit. */
  truncated: boolean;
  query: ApiEventQuery;
}

export type RegionId = "global" | "wc";

export interface ApiRegion {
  id: RegionId;
  name: string;
  subtitle: string;
  minMag: number;
  days: number;
  bbox: BBox | null;
}

export type RunKind = "feed_day" | "feed_week" | "backfill" | "details";

export interface ApiRun {
  id: number;
  kind: RunKind;
  startedAt: string;
  finishedAt: string | null;
  ok: boolean | null;
  eventsInserted: number;
  eventsUpdated: number;
  eventsSkipped: number;
  detailsFetched: number;
  mechanismsUpserted: number;
  error: string | null;
  detail: Record<string, unknown>;
}

export interface ApiStatus {
  generatedAt: string;
  events: {
    total: number;
    newest: string | null;
    oldest: string | null;
    /** Events with at least one mechanism row. */
    withMechanism: number;
  };
  mechanisms: { total: number };
  runs: Record<RunKind, ApiRun | null>;
  /** Latest finished_at of a successful feed_day or feed_week run. */
  lastSuccessfulFeed: string | null;
}

export interface ApiIssue {
  path: string;
  message: string;
}

export interface ApiError {
  error: string;
  issues?: ApiIssue[];
}

/** Preferred rows for the given events, ordered by event then kind. Empty input returns empty. */
export async function listPreferredMechanisms(
  db: Db,
  eventIds: string[],
): Promise<StoredMechanism[]> {
  if (eventIds.length === 0) return [];
  const rows = await db
    .select()
    .from(mechanisms)
    .where(and(inArray(mechanisms.eventId, eventIds), eq(mechanisms.preferred, true)))
    .orderBy(asc(mechanisms.eventId), asc(mechanisms.kind));
  return rows.map(rowToMechanism);
}

/** `listEvents` plus each event's preferred mechanisms, fetched in one query. */
export async function listEventsWithPreferred(
  db: Db,
  filter: ListEventsFilter = {},
): Promise<EventWithMechanisms[]> {
  const list = await listEvents(db, filter);
  const byEvent = new Map<string, StoredMechanism[]>();
  for (const m of await listPreferredMechanisms(
    db,
    list.map((e) => e.id),
  )) {
    const bucket = byEvent.get(m.eventId);
    if (bucket) bucket.push(m);
    else byEvent.set(m.eventId, [m]);
  }
  return list.map((e) => ({ ...e, mechanisms: byEvent.get(e.id) ?? [] }));
}

export interface CatalogStats {
  events: { total: number; newest: Date | null; oldest: Date | null; withMechanism: number };
  mechanisms: { total: number };
}

export async function catalogStats(db: Db): Promise<CatalogStats> {
  const [eventCount] = await db.select({ total: count() }).from(events);
  const [newest] = await db
    .select({ time: events.time })
    .from(events)
    .orderBy(desc(events.time))
    .limit(1);
  const [oldest] = await db
    .select({ time: events.time })
    .from(events)
    .orderBy(asc(events.time))
    .limit(1);
  const [mechCount] = await db
    .select({ total: count(), withMechanism: countDistinct(mechanisms.eventId) })
    .from(mechanisms);
  return {
    events: {
      total: eventCount?.total ?? 0,
      newest: newest?.time ?? null,
      oldest: oldest?.time ?? null,
      withMechanism: mechCount?.withMechanism ?? 0,
    },
    mechanisms: { total: mechCount?.total ?? 0 },
  };
}

export interface ListRunsFilter {
  kind?: RunKind | undefined;
  limit?: number | undefined;
}

export async function listRuns(db: Db, filter: ListRunsFilter = {}): Promise<IngestRunRow[]> {
  return db
    .select()
    .from(ingestRuns)
    .where(filter.kind ? eq(ingestRuns.kind, filter.kind) : undefined)
    .orderBy(desc(ingestRuns.startedAt), desc(ingestRuns.id))
    .limit(filter.limit ?? 50);
}

/** The most recently started run of each kind; null for a kind that never ran. */
export async function latestRuns(db: Db): Promise<Record<RunKind, IngestRunRow | null>> {
  const rows = await db
    .selectDistinctOn([ingestRuns.kind])
    .from(ingestRuns)
    .orderBy(ingestRuns.kind, desc(ingestRuns.startedAt), desc(ingestRuns.id));
  const out: Record<RunKind, IngestRunRow | null> = {
    feed_day: null,
    feed_week: null,
    backfill: null,
    details: null,
  };
  for (const row of rows) out[row.kind] = row;
  return out;
}

export async function lastSuccessfulFeedAt(db: Db): Promise<Date | null> {
  const rows = await db
    .select({ finishedAt: ingestRuns.finishedAt })
    .from(ingestRuns)
    .where(and(eq(ingestRuns.ok, true), inArray(ingestRuns.kind, ["feed_day", "feed_week"])))
    .orderBy(desc(ingestRuns.finishedAt))
    .limit(1);
  return rows[0]?.finishedAt ?? null;
}

import type { ApiError, ApiIssue } from "@tectonic/shared";
import type { FastifyInstance } from "fastify";

export class HttpError extends Error {
  readonly statusCode: number;
  readonly issues: ApiIssue[] | undefined;

  constructor(statusCode: number, message: string, issues?: ApiIssue[]) {
    super(message);
    this.name = "HttpError";
    this.statusCode = statusCode;
    this.issues = issues;
  }
}

export const badRequest = (issues: ApiIssue[]): HttpError =>
  new HttpError(400, "invalid request", issues);

export const notFound = (message = "not found"): HttpError => new HttpError(404, message);

/** One error shape for every failure: `{ error, issues? }`. */
export function registerErrorHandlers(app: FastifyInstance): void {
  app.setNotFoundHandler((_request, reply) => reply.code(404).send({ error: "not found" }));
  app.setErrorHandler((err: unknown, request, reply) => {
    if (err instanceof HttpError) {
      const body: ApiError = { error: err.message };
      if (err.issues) body.issues = err.issues;
      return reply.code(err.statusCode).send(body);
    }
    request.log.error({ err }, "unhandled error");
    return reply.code(500).send({ error: "internal error" });
  });
}

import { z } from "zod";
import type { ListEventsFilter } from "../repositories/events.js";
import { RUN_KINDS } from "../repositories/ingest-runs.js";
import { badRequest } from "./errors.js";

const DEFAULT_WINDOW_MS = 7 * 24 * 60 * 60 * 1000;

const isoDate = z
  .union([z.iso.datetime({ offset: true }), z.iso.date()])
  .transform((s) => new Date(s));

const lon = z.number().min(-180).max(180);
const lat = z.number().min(-90).max(90);

/** `minLon,minLat,maxLon,maxLat`; minLon > maxLon crosses the antimeridian. */
const bboxString = z
  .string()
  .transform((s) => s.split(",").map(Number))
  .pipe(z.tuple([lon, lat, lon, lat]))
  .transform(([minLon, minLat, maxLon, maxLat]): BBox => ({ minLon, minLat, maxLon, maxLat }))
  .refine((b) => b.minLat <= b.maxLat, { message: "minLat must not exceed maxLat" });

const eventsQuerySchema = z.object({
  since: isoDate.optional(),
  until: isoDate.optional(),
  minMag: z.coerce.number().min(-2).max(10).optional(),
  bbox: bboxString.optional(),
  limit: z.coerce.number().int().min(1).max(5000).default(500),
});

export const runsQuerySchema = z.object({
  kind: z.enum(RUN_KINDS).optional(),
  limit: z.coerce.number().int().min(1).max(500).default(50),
});

export const eventIdSchema = z.object({ id: z.string().min(1).max(64) });

export function parseOrThrow<S extends z.ZodType>(schema: S, input: unknown): z.output<S> {
  const result = schema.safeParse(input);
  if (!result.success) {
    throw badRequest(
      result.error.issues.map((i) => ({ path: i.path.map(String).join("."), message: i.message })),
    );
  }
  return result.data;
}

export interface EventsQuery {
  since: Date;
  until: Date | null;
  minMag: number | null;
  bbox: BBox | null;
  limit: number;
}

/** Parses the list query and applies defaults: since = now - 7 days. */
export function parseEventsQuery(input: unknown, now: Date): EventsQuery {
  const q = parseOrThrow(eventsQuerySchema, input);
  const since = q.since ?? new Date(now.getTime() - DEFAULT_WINDOW_MS);
  const until = q.until ?? null;
  if (until && until.getTime() < since.getTime()) {
    throw badRequest([{ path: "until", message: "until must not precede since" }]);
  }
  return { since, until, minMag: q.minMag ?? null, bbox: q.bbox ?? null, limit: q.limit };
}

export function toListFilter(q: EventsQuery): ListEventsFilter {
  const filter: ListEventsFilter = { since: q.since, limit: q.limit };
  if (q.until) filter.until = q.until;
  if (q.minMag !== null) filter.minMag = q.minMag;
  if (q.bbox) filter.bbox = q.bbox;
  return filter;
}

import { createHash } from "node:crypto";
import { readFileSync } from "node:fs";
import { join } from "node:path";
import { fileURLToPath } from "node:url";

export const GEO_NAMES = ["plates", "wc-land", "wc-states"] as const;

export interface GeoAsset {
  body: Buffer;
  etag: string;
}

export type GeoAssets = ReadonlyMap<string, GeoAsset>;

/** `apps/server/data/geo/`, resolved from `src/` under tsx and from `dist/` after build. */
export const DEFAULT_GEO_DIR = fileURLToPath(new URL("../../data/geo/", import.meta.url));

/** Reads the geo files once; the ETag is the quoted SHA-1 of the body. */
export function loadGeoAssets(dir: string = DEFAULT_GEO_DIR): GeoAssets {
  const assets = new Map<string, GeoAsset>();
  for (const name of GEO_NAMES) {
    const body = readFileSync(join(dir, `${name}.json`));
    const etag = `"${createHash("sha1").update(body).digest("hex")}"`;
    assets.set(name, { body, etag });
  }
  return assets;
}

  app.get<{ Params: { name: string } }>("/api/geo/:name", (request, reply) => {
    const asset = app.geo.get(request.params.name);
    if (!asset) throw notFound("unknown geo asset");
    void reply.header("etag", asset.etag);
    cacheFor(reply, 86_400);

import { registerErrorHandlers } from "./api/errors.js";
import type { Db } from "./db/client.js";
import { loadGeoAssets, type GeoAssets } from "./geo/assets.js";
import { geoRoutes } from "./routes/geo.js";
import { healthRoutes } from "./routes/health.js";

declare module "fastify" {
  interface FastifyInstance {
    db: Db;
    now: () => Date;
    geo: GeoAssets;
  }
}

export interface AppOptions {
  db: Db;
  logger?: FastifyServerOptions["logger"];
  /** Clock for default windows and status stamps; injectable for tests. */
  now?: () => Date;
  /** Directory holding the geo JSON files; defaults to apps/server/data/geo. */
  geoDir?: string;
}

/** Builds the Fastify app without listening, so tests can inject requests. */
export function buildApp({
  db,
  logger = false,
  now = () => new Date(),
  geoDir,

  app.decorate("geo", loadGeoAssets(geoDir));
  registerErrorHandlers(app);
  void app.register(healthRoutes);
  void app.register(geoRoutes);
  return app;
}

import type { ApiEventList } from "@tectonic/shared";
import type { FastifyPluginCallback } from "fastify";
import { cacheFor } from "../api/cache.js";
import { notFound } from "../api/errors.js";
import { eventIdSchema, parseEventsQuery, parseOrThrow, toListFilter } from "../api/schemas.js";
import { iso, isoOrNull, toApiEvent, toApiEventDetail } from "../api/serialize.js";
import { getEvent, listEventsWithPreferred } from "../repositories/events.js";

export const eventRoutes: FastifyPluginCallback = (app, _opts, done) => {
  app.get("/api/events", async (request, reply): Promise<ApiEventList> => {
    const q = parseEventsQuery(request.query, app.now());
    const rows = await listEventsWithPreferred(app.db, toListFilter(q));
    cacheFor(reply, 60);
    return {
      events: rows.map((e) => toApiEvent(e, e.mechanisms)),
      count: rows.length,
      truncated: rows.length >= q.limit,
      query: {
        since: iso(q.since),
        until: isoOrNull(q.until),
        minMag: q.minMag,
        bbox: q.bbox,
        limit: q.limit,
      },
    };
  });

  app.get("/api/events/:id", async (request, reply) => {
    const { id } = parseOrThrow(eventIdSchema, request.params);
    const event = await getEvent(app.db, id);
    if (!event) throw notFound("event not found");
    cacheFor(reply, 60);
    return toApiEventDetail(event);
  });

  done();
};

```ts
import type { ApiRun, ApiStatus } from "@tectonic/shared";
import type { FastifyPluginCallback } from "fastify";
import { cacheFor } from "../api/cache.js";
import { iso, isoOrNull, toApiRun } from "../api/serialize.js";
import type { IngestRunRow } from "../db/schema.js";
import { catalogStats } from "../repositories/events.js";
import { lastSuccessfulFeedAt, latestRuns } from "../repositories/ingest-runs.js";

const runOrNull = (r: IngestRunRow | null): ApiRun | null => (r ? toApiRun(r) : null);

export const statusRoutes: FastifyPluginCallback = (app, _opts, done) => {
  app.get("/api/status", async (_request, reply): Promise<ApiStatus> => {
    const [stats, runs, lastFeed] = await Promise.all([
      catalogStats(app.db),
      latestRuns(app.db),
      lastSuccessfulFeedAt(app.db),
    ]);
    cacheFor(reply, 15);
    return {
      generatedAt: iso(app.now()),
      events: {
        total: stats.events.total,
        newest: isoOrNull(stats.events.newest),
        oldest: isoOrNull(stats.events.oldest),
        withMechanism: stats.events.withMechanism,
      },
      mechanisms: { total: stats.mechanisms.total },
      runs: {
        feed_day: runOrNull(runs.feed_day),
        feed_week: runOrNull(runs.feed_week),
        backfill: runOrNull(runs.backfill),
        details: runOrNull(runs.details),
      },
      lastSuccessfulFeed: isoOrNull(lastFeed),
    };
  });
  done();
};

import type { ApiRun } from "@tectonic/shared";
import type { FastifyPluginCallback } from "fastify";
import { cacheFor } from "../api/cache.js";
import { parseOrThrow, runsQuerySchema } from "../api/schemas.js";
import { toApiRun } from "../api/serialize.js";
import { listRuns } from "../repositories/ingest-runs.js";

export const runRoutes: FastifyPluginCallback = (app, _opts, done) => {
  app.get("/api/runs", async (request, reply): Promise<{ runs: ApiRun[] }> => {
    const q = parseOrThrow(runsQuerySchema, request.query);
    const rows = await listRuns(app.db, { kind: q.kind, limit: q.limit });
    cacheFor(reply, 15);
    return { runs: rows.map(toApiRun) };
  });
  done();
};

## API

All routes return JSON. Bad input is 400 `{ error, issues: [{ path, message }] }`, an unknown
id or path is 404 `{ error }`, anything else is 500 `{ error }`. Successful responses carry a
short `Cache-Control`.

| Route                 | Returns                                                                                    |
| --------------------- | ------------------------------------------------------------------------------------------ |
| `GET /api/health`     | `{ status, db }`, 503 when the database does not answer                                    |
| `GET /api/regions`    | the view presets: id, name, subtitle, minMag, days, bbox                                   |
| `GET /api/events`     | `{ events, count, truncated, query }`; each event carries its preferred mechanism per kind |
| `GET /api/events/:id` | the event with every mechanism product and the raw USGS properties as `usgs`               |
| `GET /api/status`     | catalog totals, latest ingest run per kind, last successful feed time                      |
| `GET /api/runs`       | ingest run history, `?kind=feed_day                                                        | feed_week | backfill | details&limit=N` |
| `GET /api/geo/:name`  | `plates`, `wc-land`, `wc-states` from `apps/server/data/geo`, with an ETag                 |

`GET /api/events` filters: `since` and `until` (ISO 8601 date-time or `YYYY-MM-DD`; `since`
defaults to seven days ago), `minMag`, `bbox=minLon,minLat,maxLon,maxLat` (minLon > maxLon
crosses the antimeridian), `limit` (default 500, max 5000). Dates in responses are ISO 8601 UTC.
Types for every response are in `packages/shared/src/api.ts`.
```
