```ts
import type { Axis, NodalPlane } from "./planes.js";

const RAD = Math.PI / 180;

export type Vec3 = [number, number, number];
/** Symmetric 3×3 tensor in the north-east-down frame, unit scalar moment. */
export type Tensor = [Vec3, Vec3, Vec3];

/** Unit vector in the north-east-down frame. */
export function axisVector(a: Axis): Vec3 {
  const az = a.azimuth * RAD;
  const pl = a.plunge * RAD;
  return [Math.cos(pl) * Math.cos(az), Math.cos(pl) * Math.sin(az), Math.sin(pl)];
}

/** Double-couple tensor from strike, dip and rake (Aki & Richards, NED). */
export function tensorFromPlane(p: NodalPlane): Tensor {
  const s = p.strike * RAD;
  const d = p.dip * RAD;
  const r = p.rake * RAD;
  const sd = Math.sin(d);
  const cd = Math.cos(d);
  const s2d = Math.sin(2 * d);
  const c2d = Math.cos(2 * d);
  const sr = Math.sin(r);
  const cr = Math.cos(r);
  const ss = Math.sin(s);
  const cs = Math.cos(s);
  const s2s = Math.sin(2 * s);
  const c2s = Math.cos(2 * s);
  const xx = -(sd * cr * s2s + s2d * sr * ss * ss);
  const xy = sd * cr * c2s + 0.5 * s2d * sr * s2s;
  const xz = -(cd * cr * cs + c2d * sr * ss);
  const yy = sd * cr * s2s - s2d * sr * cs * cs;
  const yz = -(cd * cr * ss - c2d * sr * cs);
  const zz = s2d * sr;
  return [
    [xx, xy, xz],
    [xy, yy, yz],
    [xz, yz, zz],
  ];
}

/** Double-couple tensor T⊗T − P⊗P from the tension and pressure axes. */
export function tensorFromAxes(t: Axis, p: Axis): Tensor {
  const tv = axisVector(t);
  const pv = axisVector(p);
  const e = (i: 0 | 1 | 2, j: 0 | 1 | 2): number => tv[i] * tv[j] - pv[i] * pv[j];
  return [
    [e(0, 0), e(0, 1), e(0, 2)],
    [e(1, 0), e(1, 1), e(1, 2)],
    [e(2, 0), e(2, 1), e(2, 2)],
  ];
}

/** vᵀMv: positive in compressional quadrants, negative in dilatational ones. */
export function radiation(m: Tensor, v: Vec3): number {
  const [a, b, c] = v;
  return (
    a * (m[0][0] * a + m[0][1] * b + m[0][2] * c) +
    b * (m[1][0] * a + m[1][1] * b + m[1][2] * c) +
    c * (m[2][0] * a + m[2][1] * b + m[2][2] * c)
  );
}

/** T and P axes of a tensor by a 2° grid search over the lower hemisphere. */
export function axesFromTensor(m: Tensor): { t: Axis; p: Axis } {
  let t: Axis = { azimuth: 0, plunge: 0 };
  let p: Axis = { azimuth: 0, plunge: 0 };
  let tMax = -Infinity;
  let pMin = Infinity;
  for (let az = 0; az < 360; az += 2) {
    for (let pl = 0; pl <= 90; pl += 2) {
      const a = { azimuth: az, plunge: pl };
      const r = radiation(m, axisVector(a));
      if (r > tMax) {
        tMax = r;
        t = a;
      }
      if (r < pMin) {
        pMin = r;
        p = a;
      }
    }
  }
  return { t, p };
}

/**
 * Lower-hemisphere equal-area inverse projection: the unit direction (NED)
 * plotted at (x east, y north) in the unit disc, or null outside the disc.
 */
export function directionAt(x: number, y: number): Vec3 | null {
  const rr = x * x + y * y;
  if (rr > 1) return null;
  const inc = 2 * Math.asin(Math.sqrt(rr) / Math.SQRT2);
  const az = Math.atan2(x, y);
  return [Math.sin(inc) * Math.cos(az), Math.sin(inc) * Math.sin(az), Math.cos(inc)];
}

/** Where an axis plots in the unit disc, as [east, north]. */
export function projectAxis(a: Axis): [number, number] {
  const inc = (90 - a.plunge) * RAD;
  const r = Math.SQRT2 * Math.sin(inc / 2);
  const az = a.azimuth * RAD;
  return [r * Math.sin(az), r * Math.cos(az)];
}
```

```ts
import type { NodalPlane } from "./planes.js";

/**
 * Faulting style from the rake of a nodal plane, with the prototype's
 * thresholds: reverse 30..150, normal −150..−30, otherwise strike-slip;
 * "Thrust" for a reverse plane dipping 35° or less; "Oblique" when the rake
 * sits between the pure cases.
 */
export function faultingStyle(p: NodalPlane): string {
  const r = p.rake;
  const a = Math.abs(r);
  let style: string;
  if (r >= 30 && r <= 150) style = "Reverse";
  else if (r <= -30 && r >= -150) style = "Normal";
  else style = "Strike-slip";
  const oblique = (a > 20 && a < 70) || (a > 110 && a < 160);
  if (style === "Reverse" && p.dip <= 35 && !oblique) style = "Thrust";
  return (oblique ? "Oblique " : "") + style;
}

/** Sense of slip on a strike-slip plane. */
export function strikeSlipSense(p: NodalPlane): "left-lateral" | "right-lateral" {
  return Math.abs(p.rake) < 90 ? "left-lateral" : "right-lateral";
}
```

```ts
import type { Mechanism } from "../index.js";
import { planesFromAxes, type Axis, type NodalPlane } from "./planes.js";
import { axesFromTensor, tensorFromAxes, tensorFromPlane, type Tensor } from "./tensor.js";

/** The geometry fields shared by `Mechanism` and `ApiMechanism`. */
export type MechanismGeometry = Pick<
  Mechanism,
  | "strike1"
  | "dip1"
  | "rake1"
  | "strike2"
  | "dip2"
  | "rake2"
  | "tAzimuth"
  | "tPlunge"
  | "pAzimuth"
  | "pPlunge"
>;

export interface Axes {
  t: Axis;
  p: Axis;
}

export interface SolutionPlanes {
  planes: [NodalPlane, NodalPlane];
  /** True when the planes were derived from the T and P axes. */
  reconstructed: boolean;
}

export function storedAxes(m: MechanismGeometry): Axes | null {
  if (m.tAzimuth === null || m.tPlunge === null || m.pAzimuth === null || m.pPlunge === null) {
    return null;
  }
  return {
    t: { azimuth: m.tAzimuth, plunge: m.tPlunge },
    p: { azimuth: m.pAzimuth, plunge: m.pPlunge },
  };
}

export function storedPlanes(m: MechanismGeometry): [NodalPlane, NodalPlane] | null {
  if (
    m.strike1 === null ||
    m.dip1 === null ||
    m.rake1 === null ||
    m.strike2 === null ||
    m.dip2 === null ||
    m.rake2 === null
  ) {
    return null;
  }
  return [
    { strike: m.strike1, dip: m.dip1, rake: m.rake1 },
    { strike: m.strike2, dip: m.dip2, rake: m.rake2 },
  ];
}

/** Stored planes, else planes reconstructed from the axes, else null. */
export function planesOf(m: MechanismGeometry): SolutionPlanes | null {
  const planes = storedPlanes(m);
  if (planes) return { planes, reconstructed: false };
  const axes = storedAxes(m);
  if (axes === null) return null;
  return { planes: planesFromAxes(axes.t, axes.p), reconstructed: true };
}

/** Tensor from plane 1, else from the axes, else null. */
export function tensorOf(m: MechanismGeometry): Tensor | null {
  const planes = storedPlanes(m);
  if (planes) return tensorFromPlane(planes[0]);
  const axes = storedAxes(m);
  return axes === null ? null : tensorFromAxes(axes.t, axes.p);
}

/** Stored axes, else axes found on the tensor, else null. */
export function axesOf(m: MechanismGeometry): Axes | null {
  const axes = storedAxes(m);
  if (axes) return axes;
  const tensor = tensorOf(m);
  return tensor === null ? null : axesFromTensor(tensor);
}
```

```ts
import type {
  ApiError,
  ApiEventDetail,
  ApiEventList,
  ApiRegion,
  ApiStatus,
} from "@tectonic/shared";

export interface EventsQuery {
  since: string;
  until?: string;
  minMag?: number;
  bbox?: string;
  limit?: number;
}

export type GeoName = "plates" | "wc-land" | "wc-states";

export interface ApiClient {
  regions(): Promise<ApiRegion[]>;
  events(query: EventsQuery): Promise<ApiEventList>;
  event(id: string): Promise<ApiEventDetail>;
  status(): Promise<ApiStatus>;
  geo(name: GeoName): Promise<unknown>;
}

export class ApiRequestError extends Error {
  constructor(
    readonly status: number,
    message: string,
  ) {
    super(message);
    this.name = "ApiRequestError";
  }
}

export type FetchLike = (input: string, init?: RequestInit) => Promise<Response>;

export function errorMessage(e: unknown): string {
  return e instanceof Error ? e.message : String(e);
}

function toParams(q: EventsQuery): string {
  const p = new URLSearchParams();
  p.set("since", q.since);
  if (q.until !== undefined) p.set("until", q.until);
  if (q.minMag !== undefined) p.set("minMag", String(q.minMag));
  if (q.bbox !== undefined) p.set("bbox", q.bbox);
  if (q.limit !== undefined) p.set("limit", String(q.limit));
  return p.toString();
}

export function createApiClient(fetchFn: FetchLike, base = "/api"): ApiClient {
  async function get<T>(path: string): Promise<T> {
    const res = await fetchFn(base + path, { headers: { accept: "application/json" } });
    if (!res.ok) {
      let message = `HTTP ${String(res.status)}`;
      try {
        const body = (await res.json()) as Partial<ApiError> | null;
        if (typeof body?.error === "string") message = body.error;
      } catch {
        // body was not JSON; keep the status message
      }
      throw new ApiRequestError(res.status, message);
    }
    return (await res.json()) as T;
  }
  return {
```

```ts
export interface Clock {
  now: Ref<Date>;
  start(): void;
  stop(): void;
}

/** A reactive "now" that ticks on an interval so "ago" labels stay current. */
export function createClock(intervalMs = 30_000, read: () => Date = () => new Date()): Clock {
  const now = ref(read());
  let timer: ReturnType<typeof setInterval> | null = null;
  return {
    now,
    start() {
      if (timer !== null) return;
      timer = setInterval(() => {
        now.value = read();
      }, intervalMs);
    },
    stop() {
      if (timer === null) return;
      clearInterval(timer);
      timer = null;
    },
  };
}
```

```ts
import { computed, ref, watchEffect, type ComputedRef, type Ref } from "vue";

export type ThemeMode = "auto" | "light" | "dark";

const KEY = "td-theme";
const NEXT: Record<ThemeMode, ThemeMode> = { auto: "light", light: "dark", dark: "auto" };

type StorageLike = Pick<Storage, "getItem" | "setItem">;
interface MediaLike {
  matches: boolean;
  addEventListener(type: "change", listener: (e: { matches: boolean }) => void): void;
}

export interface ThemeOptions {
  storage?: StorageLike | null;
  media?: MediaLike;
  root?: HTMLElement;
}

export interface Theme {
  mode: Ref<ThemeMode>;
  resolved: ComputedRef<"light" | "dark">;
  set(mode: ThemeMode): void;
  cycle(): void;
}

function isMode(v: string | null): v is ThemeMode {
  return v === "auto" || v === "light" || v === "dark";
}

/** auto / light / dark, persisted, applied as `data-theme` on the root element. */
export function createTheme(opts: ThemeOptions = {}): Theme {
  const storage = opts.storage === undefined ? localStorage : opts.storage;
  const media = opts.media ?? matchMedia("(prefers-color-scheme: dark)");
  const root = opts.root ?? document.documentElement;
  const stored = storage?.getItem(KEY) ?? null;
  const mode = ref<ThemeMode>(isMode(stored) ? stored : "auto");
  const systemDark = ref(media.matches);
  media.addEventListener("change", (e) => {
    systemDark.value = e.matches;
  });
  const resolved = computed(() =>
    mode.value === "auto" ? (systemDark.value ? "dark" : "light") : mode.value,
  );
  watchEffect(() => {
    if (mode.value === "auto") delete root.dataset.theme;
    else root.dataset.theme = mode.value;
  });
  function set(m: ThemeMode): void {
    mode.value = m;
    try {
      storage?.setItem(KEY, m);
    } catch {
      // storage full or disabled; the choice still applies for this page
    }
  }
  return {
    mode,
    resolved,
    set,
    cycle: () => {
      set(NEXT[mode.value]);
    },
  };
}
```

```ts
import type { ApiEvent } from "@tectonic/shared";

const RAD = Math.PI / 180;
const SEQUENCE_KM = 150;
const MAINSHOCK_MAG = 6.5;

interface LatLon {
  latitude: number;
  longitude: number;
}

/** Great-circle distance by the spherical law of cosines, R = 6371 km. */
export function distanceKm(a: LatLon, b: LatLon): number {
  const p1 = a.latitude * RAD;
  const p2 = b.latitude * RAD;
  const dl = (b.longitude - a.longitude) * RAD;
  const c = Math.sin(p1) * Math.sin(p2) + Math.cos(p1) * Math.cos(p2) * Math.cos(dl);
  return 6371 * Math.acos(Math.min(1, c));
}

/** A mainshock's followers. Only returned when there is at least one. */
export interface Sequence {
  count: number;
  largest: ApiEvent;
  spanHours: number;
}

const t = (e: ApiEvent): number => Date.parse(e.time);
const magOf = (e: ApiEvent): number => e.mag ?? -Infinity;

/** The M6.5+ event this one follows within 150 km, if any. */
export function parentOf(e: ApiEvent, list: ApiEvent[]): ApiEvent | null {
  return (
    list.find((o) => magOf(o) >= MAINSHOCK_MAG && t(o) < t(e) && distanceKm(o, e) <= SEQUENCE_KM) ??
    null
  );
}

/**
 * Later events within 150 km of an M6.5+ mainshock that is not itself inside
 * another sequence; null when the event is not a mainshock or has no followers.
 */
export function sequenceOf(e: ApiEvent, list: ApiEvent[], now: Date): Sequence | null {
  if (magOf(e) < MAINSHOCK_MAG || parentOf(e, list) !== null) return null;
  const after = list.filter((o) => t(o) > t(e) && distanceKm(e, o) <= SEQUENCE_KM);
  const first = after[0];
  if (first === undefined) return null;
  const largest = after.reduce((a, o) => (magOf(o) > magOf(a) ? o : a), first);
  return { count: after.length, largest, spanHours: (now.getTime() - t(e)) / 36e5 };
}
```

```ts
import { planesOf, type ApiEvent, type ApiMechanism, type ApiRegion } from "@tectonic/shared";

export interface BValue {
  b: number;
  sigma: number;
  n: number;
}

/** Aki (1965) maximum-likelihood b with the Shi & Bolt (1982) uncertainty. */
export function bValue(mags: number[], mc: number): BValue | null {
  const ms = mags.filter((m) => m >= mc - 1e-9);
  const n = ms.length;
  if (n < 2) return null;
  const mean = ms.reduce((a, m) => a + m, 0) / n;
  const b = Math.LOG10E / (mean - (mc - 0.05));
  const ss = ms.reduce((a, m) => a + (m - mean) ** 2, 0);
  return { b, sigma: 2.3 * b * b * Math.sqrt(ss / (n * (n - 1))), n };
}

export interface MagBin {
  m: number;
  cumulative: number;
  incremental: number;
}

/** 0.1-magnitude bins from `mc` to the largest magnitude. */
export function magnitudeBins(mags: number[], mc: number): MagBin[] {
  if (mags.length === 0) return [];
  const max = Math.max(...mags);
  const bins: MagBin[] = [];
  for (let m = mc; m <= max + 1e-9; m += 0.1) {
    const v = Number(m.toFixed(1));
    bins.push({
      m: v,
      cumulative: mags.filter((x) => x >= v - 1e-9).length,
      incremental: mags.filter((x) => Math.abs(x - v) < 0.05).length,
    });
  }
  return bins;
}

const magOf = (e: ApiEvent): number => e.mag ?? -Infinity;
const depthOf = (e: ApiEvent): number => e.depthKm ?? -Infinity;
const drawable = (m: ApiMechanism): boolean => planesOf(m) !== null;

/** The solution to draw: preferred mt, preferred fm, then any drawable product. */
export function preferredMechanism(e: ApiEvent): ApiMechanism | null {
  const ms = e.mechanisms.filter(drawable);
  return (
    ms.find((m) => m.kind === "mt" && m.preferred) ??
    ms.find((m) => m.kind === "fm" && m.preferred) ??
    ms[0] ??
    null
  );
}

export function hasSolution(e: ApiEvent): boolean {
  return preferredMechanism(e) !== null;
}

export interface Summary {
  count: number;
  last24h: number;
  largest: ApiEvent | null;
  deepest: ApiEvent | null;
  withSolution: number;
}

export function summary(events: ApiEvent[], now: Date): Summary {
  const dayAgo = now.getTime() - 864e5;
  let largest: ApiEvent | null = null;
  let deepest: ApiEvent | null = null;
  let last24h = 0;
  let withSolution = 0;
  for (const e of events) {
    if (Date.parse(e.time) >= dayAgo) last24h += 1;
    if (hasSolution(e)) withSolution += 1;
    if (e.mag !== null && (largest === null || e.mag > magOf(largest))) largest = e;
    if (e.depthKm !== null && (deepest === null || e.depthKm > depthOf(deepest))) deepest = e;
  }
  return { count: events.length, last24h, largest, deepest, withSolution };
}

const byMagDesc = (a: ApiEvent, b: ApiEvent): number => magOf(b) - magOf(a);

export function majorEvents(events: ApiEvent[], region: ApiRegion): ApiEvent[] {
  if (region.id === "global") {
    return events.filter((e) => magOf(e) >= 6 || e.alert !== null).sort(byMagDesc);
  }
  return [...events].sort(byMagDesc).slice(0, 4);
}

export function majorLabel(region: ApiRegion): string {
  return region.id === "global"
    ? "Major events · M6+ or a PAGER alert"
    : "Largest events in the region";
}
```

```ts
  axesFromTensor,
  directionAt,
  projectAxis,
  radiation,
  storedAxes,
  tensorOf,
  type MechanismGeometry,
} from "@tectonic/shared";
import { color as parseColor } from "d3-color";

export interface BeachballColors {
  fill: string;
  tension: string;
  ink: string;
}

interface Rgb {
  r: number;
  g: number;
  b: number;
  css: string;
}

const cache = new Map<string, string>();

export function clearBeachballCache(): void {
  cache.clear();
}

function rgb(s: string): Rgb {
  const c = parseColor(s);
  if (c === null) return { r: 0, g: 0, b: 0, css: "#000" };
  const o = c.rgb();
  return { r: o.r, g: o.g, b: o.b, css: o.formatRgb() };
}

/**
 * A lower-hemisphere equal-area beachball as a PNG data URL, rendered at
 * twice `px` for sharpness. Compressional quadrants take `fill`, dilatational
 * take `tension`; the rim and the T/P markers take `ink`. Empty string when
 * the mechanism has no tensor or the canvas has no 2D context.
 */
export function beachballUrl(
  m: MechanismGeometry,
  px: number,
  colors: BeachballColors,
  showAxes: boolean,
): string {
  const tensor = tensorOf(m);
  if (tensor === null) return "";
  const key = [
    m.strike1,
    m.dip1,
    m.rake1,
    m.tAzimuth,
    m.tPlunge,
    m.pAzimuth,
    m.pPlunge,
    px,
    colors.fill,
    colors.tension,
    colors.ink,
    showAxes,
  ].join("|");
  const hit = cache.get(key);
  if (hit !== undefined) return hit;

  const scale = 2;
  const n = px * scale;
  const canvas = document.createElement("canvas");
  canvas.width = n;
  canvas.height = n;
  const g = canvas.getContext("2d");
  if (g === null) return "";
  const fill = rgb(colors.fill);
  const tension = rgb(colors.tension);
  const ink = rgb(colors.ink);
  const r0 = n / 2 - 1.5 * scale;

  const img = g.createImageData(n, n);
  for (let y = 0; y < n; y++) {
    for (let x = 0; x < n; x++) {
      const v = directionAt((x + 0.5 - n / 2) / r0, (n / 2 - y - 0.5) / r0);
      if (v === null) continue;
      const c = radiation(tensor, v) > 0 ? fill : tension;
      const k = (y * n + x) * 4;
      img.data[k] = c.r;
      img.data[k + 1] = c.g;
      img.data[k + 2] = c.b;
      img.data[k + 3] = 255;
    }
  }
  g.putImageData(img, 0, 0);
  g.beginPath();
  g.arc(n / 2, n / 2, r0, 0, 2 * Math.PI);
  g.lineWidth = Math.max(1.2, n / 70);
  g.strokeStyle = ink.css;
  g.stroke();

  if (showAxes) {
    const axes = storedAxes(m) ?? axesFromTensor(tensor);
    g.font = `600 ${String(n / 11)}px "IBM Plex Mono", monospace`;
    g.textAlign = "center";
    g.textBaseline = "middle";
    for (const [label, axis] of [
      ["T", axes.t],
      ["P", axes.p],
    ] as const) {
      const [east, north] = projectAxis(axis);
      const cx = n / 2 + east * r0;
      const cy = n / 2 - north * r0;
      g.beginPath();
      g.arc(cx, cy, n / 15, 0, 2 * Math.PI);
      g.fillStyle = ink.css;
      g.fill();
      g.fillStyle = tension.css;
      g.fillText(label, cx, cy + 1);
    }
  }
  const url = canvas.toDataURL();
  cache.set(key, url);
  return url;
}
```

```ts
import { computed, ref, shallowRef, type ComputedRef, type Ref, type ShallowRef } from "vue";
import { errorMessage, type ApiClient, type EventsQuery } from "./api.js";

export const EVENT_LIMIT = 2000;

export function eventsQueryFor(region: ApiRegion, now: Date): EventsQuery {
  const q: EventsQuery = {
    since: new Date(now.getTime() - region.days * 864e5).toISOString(),
    minMag: region.minMag,
    limit: EVENT_LIMIT,
  };
  if (region.bbox !== null) {
    const b = region.bbox;
    q.bbox = `${String(b.minLon)},${String(b.minLat)},${String(b.maxLon)},${String(b.maxLat)}`;
  }
  return q;
}

export interface Catalog {
  regions: ShallowRef<ApiRegion[]>;
  region: ComputedRef<ApiRegion | null>;
  events: ShallowRef<ApiEvent[]>;
  truncated: Ref<boolean>;
  loading: Ref<boolean>;
  error: Ref<string | null>;
  selectedId: Ref<string | null>;
  selected: ComputedRef<ApiEvent | null>;
  /** The active region; throws when none is set (components below the region guard never see that). */
  requireRegion(): ApiRegion;
  /** Loads the region presets. */
  init(): Promise<void>;
  /** Switches region and loads its events; false when the id is unknown. */
  setRegion(id: string): Promise<boolean>;
  select(id: string | null): void;
  /** Reloads the current region's events, keeping the selection. */
  refetch(): Promise<void>;
}

export function createCatalog(api: ApiClient, now: () => Date): Catalog {
  const regions = shallowRef<ApiRegion[]>([]);
  const regionId = ref<string | null>(null);
  const events = shallowRef<ApiEvent[]>([]);
  const truncated = ref(false);
  const loading = ref(false);
  const error = ref<string | null>(null);
  const selectedId = ref<string | null>(null);
  const region = computed(() => regions.value.find((r) => r.id === regionId.value) ?? null);
  const selected = computed(() => events.value.find((e) => e.id === selectedId.value) ?? null);
  let seq = 0;

  async function load(r: ApiRegion): Promise<void> {
    const token = ++seq;
    loading.value = true;
    error.value = null;
    try {
      const list = await api.events(eventsQueryFor(r, now()));
      if (token !== seq) return;
      events.value = list.events;
      truncated.value = list.truncated;
    } catch (e) {
      if (token !== seq) return;
      error.value = errorMessage(e);
    } finally {
      if (token === seq) loading.value = false;
    }
  }

  return {
    regions,
    region,
    events,
    truncated,
    loading,
    error,
    selectedId,
    selected,
    requireRegion() {
      const r = region.value;
      if (r === null) throw new Error("no region selected");
      return r;
    },
    async init() {
      try {
        regions.value = await api.regions();
      } catch (e) {
        error.value = errorMessage(e);
      }
    },
    async setRegion(id) {
      const r = regions.value.find((x) => x.id === id);
      if (r === undefined) return false;
      if (regionId.value === id) return true;
      regionId.value = id;
      selectedId.value = null;
      events.value = [];
      truncated.value = false;
      await load(r);
      return true;
    },
    select(id) {
      selectedId.value = id;
    },
    async refetch() {
      const r = region.value;
      if (r !== null) await load(r);
    },
  };
}
```

```ts
import type { ApiStatus } from "@tectonic/shared";
import { ref, type Ref } from "vue";
import { errorMessage, type ApiClient } from "./api.js";

type DocLike = Pick<Document, "hidden" | "addEventListener" | "removeEventListener">;

export interface StatusPollOptions {
  intervalMs?: number;
  doc?: DocLike;
}

export interface StatusPoll {
  status: Ref<ApiStatus | null>;
  error: Ref<string | null>;
  /** When the current run of failures began; null while healthy. */
  errorSince: Ref<Date | null>;
  fetchedAt: Ref<Date | null>;
  poll(): Promise<void>;
  start(): void;
  stop(): void;
}

/**
 * Polls `GET /api/status`. `onFeedChange` fires when `lastSuccessfulFeed`
 * differs from the previous non-null value, which is the signal to refetch
 * events. Polling pauses while the document is hidden.
 */
export function createStatusPoll(
  api: ApiClient,
  now: () => Date,
  onFeedChange: () => void,
  opts: StatusPollOptions = {},
): StatusPoll {
  const intervalMs = opts.intervalMs ?? 60_000;
  const doc: DocLike = opts.doc ?? document;
  const status = ref<ApiStatus | null>(null);
  const error = ref<string | null>(null);
  const errorSince = ref<Date | null>(null);
  const fetchedAt = ref<Date | null>(null);
  let lastFeed: string | null = null;
  let timer: ReturnType<typeof setInterval> | null = null;
  let started = false;

  async function poll(): Promise<void> {
    try {
      const s = await api.status();
      status.value = s;
      error.value = null;
      errorSince.value = null;
      if (lastFeed !== null && s.lastSuccessfulFeed !== null && s.lastSuccessfulFeed !== lastFeed) {
        onFeedChange();
      }
      lastFeed = s.lastSuccessfulFeed ?? lastFeed;
    } catch (e) {
      error.value = errorMessage(e);
      errorSince.value ??= now();
    }
    fetchedAt.value = now();
  }

  function startTimer(): void {
    if (timer === null) timer = setInterval(() => void poll(), intervalMs);
  }
  function stopTimer(): void {
    if (timer !== null) {
      clearInterval(timer);
      timer = null;
    }
  }
  function onVisibility(): void {
    if (doc.hidden) {
      stopTimer();
    } else {
      void poll();
      startTimer();
    }
  }

  return {
    status,
    error,
    errorSince,
    fetchedAt,
    poll,
    start() {
      if (started) return;
      started = true;
      void poll();
      startTimer();
      doc.addEventListener("visibilitychange", onVisibility);
    },
    stop() {
      if (!started) return;
      started = false;
      stopTimer();
      doc.removeEventListener("visibilitychange", onVisibility);
    },
  };
}
```

```ts
export function markerRadius(mag: number, minMag: number, global: boolean): number {
```

```ts
export interface Tick {
  pos: number;
  label: string;
}

/** The subset of a d3 continuous scale that axes need. */
export interface AxisScale<T> {
  (value: T): number;
  ticks(count?: number): T[];
  tickFormat(count?: number, specifier?: string): (value: T) => string;
}

export function ticksOf<T>(scale: AxisScale<T>, count: number, specifier?: string): Tick[] {
  const format = scale.tickFormat(count, specifier);
  return scale.ticks(count).map((v) => ({ pos: scale(v), label: format(v) }));
}

/** SVG path data for a step-after line through the points. */
export function stepPath(points: { x: number; y: number }[]): string {
  return points
    .map((p, i) => (i === 0 ? `M${String(p.x)},${String(p.y)}` : `H${String(p.x)}V${String(p.y)}`))
    .join("");
}
```

```vue
const sortDir = ref<1 | -1>(-1);

function sortBy(k: SortKey): void {
  if (sortKey.value === k) {
    sortDir.value = sortDir.value === 1 ? -1 : 1;
  } else {
    sortKey.value = k;
    sortDir.value = k === "place" ? 1 : -1;
  }
}
function value(e: ApiEvent): string | number {
  const v = e[sortKey.value];
  if (v !== null) return v;
  return sortKey.value === "place" ? "" : -Infinity;
}

);
```

```vue
  const token = ++seq;
  loading.value = true;
  notFound.value = false;
  error.value = null;

    const d = await api.event(id);
```
