# Ingest Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fill and keep current the `events` and `mechanisms` tables from USGS: poll the 2.5_day and 2.5_week feeds, fetch detail documents for events that advertise focal mechanisms, backfill 90 days on demand, and record every run.

**Architecture:** Pipeline modules under `apps/server/src/usgs` (HTTP client, zod schemas, pure parsers) and `apps/server/src/ingest` (sync, backfill, scheduler), all driven by an `IngestContext { db, usgs, log, now }` so tests inject a stub client and clock. Nodal-plane reconstruction lives in `packages/shared`, ported from `quake/bb.js`. The scheduler runs inside the server behind `INGEST_ENABLED`; a CLI covers one-off runs.

**Tech Stack:** Additions to Foundation: `pino` 10.4.0 (direct dep for the CLI logger), vitest in `packages/shared`. Everything else is already installed.

```ts
const RAD = Math.PI / 180;

export interface Axis {
  /** Degrees clockwise from north. */
  azimuth: number;
  /** Degrees below horizontal. */
  plunge: number;
}

export interface NodalPlane {
  strike: number;
  dip: number;
  rake: number;
}

type Vec3 = [number, number, number];

/** Unit vector in the north-east-down frame. */
function axisVector(a: Axis): Vec3 {
  const az = a.azimuth * RAD;
  const pl = a.plunge * RAD;
  return [Math.cos(pl) * Math.cos(az), Math.cos(pl) * Math.sin(az), Math.sin(pl)];
}

/** Strike, dip and rake of the plane with normal `n` and slip `d` (Aki & Richards). */
function planeFrom(normal: Vec3, slip: Vec3): NodalPlane {
  let n = normal;
  let d = slip;
  if (n[2] > 0) {
    n = [-n[0], -n[1], -n[2]];
    d = [-d[0], -d[1], -d[2]];
  }
  const dip = Math.acos(-n[2]);
  let strike = Math.atan2(-n[0], n[1]);
  if (strike < 0) strike += 2 * Math.PI;
  const sinDip = Math.sin(dip);
  const sl = sinDip === 0 ? 0 : -d[2] / sinDip;
  const cl = d[0] * Math.cos(strike) + d[1] * Math.sin(strike);
  return { strike: strike / RAD, dip: dip / RAD, rake: Math.atan2(sl, cl) / RAD };
}

/**
 * Both nodal planes of a double couple from its T (tension) and P (pressure)
 * axes. Normal and slip are n = (T + P) / sqrt2 and d = (T - P) / sqrt2; the
 * auxiliary plane swaps them. Ported from the verified prototype math.
 */
export function planesFromAxes(t: Axis, p: Axis): [NodalPlane, NodalPlane] {
  const tv = axisVector(t);
  const pv = axisVector(p);
  const n: Vec3 = [
    (tv[0] + pv[0]) / Math.SQRT2,
    (tv[1] + pv[1]) / Math.SQRT2,
    (tv[2] + pv[2]) / Math.SQRT2,
  ];
  const d: Vec3 = [
    (tv[0] - pv[0]) / Math.SQRT2,
    (tv[1] - pv[1]) / Math.SQRT2,
    (tv[2] - pv[2]) / Math.SQRT2,
  ];
  return [planeFrom(n, d), planeFrom(d, n)];
}
```

```ts
const MECHANISM_TYPES = ["moment-tensor", "focal-mechanism"];

/** Events advertising a mechanism product whose detail has not been fetched since their last revision. */
export async function listDetailCandidates(
  db: Db,
  limit: number,
): Promise<{ id: string; updated: Date }[]> {
  return db
    .select({ id: events.id, updated: events.updated })
    .from(events)
    .where(
      and(
        arrayOverlaps(events.types, MECHANISM_TYPES),
        or(isNull(events.detailCheckedUpdated), lt(events.detailCheckedUpdated, events.updated)),
      ),
    )
    .orderBy(desc(events.time), desc(events.id))
    .limit(limit);
}

export async function markDetailChecked(db: Db, id: string, updated: Date): Promise<void> {
  await db.update(events).set({ detailCheckedUpdated: updated }).where(eq(events.id, id));
}
```

```ts
import { eq } from "drizzle-orm";
import type { Db } from "../db/client.js";
import { ingestRuns, type IngestRunRow } from "../db/schema.js";

export type RunKind = IngestRunRow["kind"];

export interface RunResult {
  ok: boolean;
  error?: string;
  eventsInserted?: number;
  eventsUpdated?: number;
  eventsSkipped?: number;
  detailsFetched?: number;
  mechanismsUpserted?: number;
}

export async function startRun(
  db: Db,
  kind: RunKind,
  detail: Record<string, unknown> = {},
  startedAt: Date = new Date(),
): Promise<number> {
  const rows = await db
    .insert(ingestRuns)
    .values({ kind, startedAt, detail })
    .returning({ id: ingestRuns.id });
  const first = rows[0];
  if (!first) throw new Error("insert into ingest_runs returned no row");
  return first.id;
}

export async function finishRun(
  db: Db,
  id: number,
  result: RunResult,
  finishedAt: Date = new Date(),
): Promise<void> {
  await db
    .update(ingestRuns)
    .set({
      finishedAt,
      ok: result.ok,
      error: result.error ?? null,
      eventsInserted: result.eventsInserted ?? 0,
      eventsUpdated: result.eventsUpdated ?? 0,
      eventsSkipped: result.eventsSkipped ?? 0,
      detailsFetched: result.detailsFetched ?? 0,
      mechanismsUpserted: result.mechanismsUpserted ?? 0,
    })
    .where(eq(ingestRuns.id, id));
}

export async function getRun(db: Db, id: number): Promise<IngestRunRow | null> {
  const rows = await db.select().from(ingestRuns).where(eq(ingestRuns.id, id)).limit(1);
  return rows[0] ?? null;
}
```

```ts
import type { z } from "zod";
import {
  detailDocumentSchema,
  feedDocumentSchema,
  type DetailDocument,
  type FeedDocument,
} from "./schema.js";

export type FeedKind = "2.5_day" | "2.5_week";

export interface QueryParams {
  starttime: Date;
  endtime: Date;
  minmagnitude: number;
  limit: number;
}

export interface UsgsClientOptions {
  fetch?: typeof fetch;
  baseUrl: string;
  userAgent: string;
  timeoutMs?: number;
}

export interface UsgsClient {
  fetchFeed: (kind: FeedKind) => Promise<FeedDocument>;
  fetchDetail: (eventId: string) => Promise<DetailDocument>;
  fetchQuery: (params: QueryParams) => Promise<FeedDocument>;
}

export class UsgsHttpError extends Error {
  override name = "UsgsHttpError";
  constructor(
    public readonly status: number,
    public readonly url: string,
  ) {
    super(`USGS responded ${String(status)} for ${url}`);
  }
}

export function createUsgsClient(opts: UsgsClientOptions): UsgsClient {
  const doFetch = opts.fetch ?? fetch;
  const timeoutMs = opts.timeoutMs ?? 15_000;
  const base = opts.baseUrl.replace(/\/$/, "");

  async function getJson<T>(url: string, schema: z.ZodType<T>): Promise<T> {
    const controller = new AbortController();
    const timer = setTimeout(() => {
      controller.abort();
    }, timeoutMs);
    try {
      const res = await doFetch(url, {
        headers: { "user-agent": opts.userAgent, accept: "application/json" },
        signal: controller.signal,
      });
      if (!res.ok) throw new UsgsHttpError(res.status, url);
      const body: unknown = await res.json();
      return schema.parse(body);
    } finally {
      clearTimeout(timer);
    }
  }

  return {
    fetchFeed: (kind) =>
      getJson(`${base}/earthquakes/feed/v1.0/summary/${kind}.geojson`, feedDocumentSchema),
    fetchDetail: (eventId) =>
      getJson(
        `${base}/fdsnws/event/1/query?${new URLSearchParams({ eventid: eventId, format: "geojson" }).toString()}`,
        detailDocumentSchema,
      ),
    fetchQuery: (p) =>
      getJson(
        `${base}/fdsnws/event/1/query?${new URLSearchParams({
          format: "geojson",
          starttime: p.starttime.toISOString(),
          endtime: p.endtime.toISOString(),
          minmagnitude: String(p.minmagnitude),
          orderby: "time-asc",
          limit: String(p.limit),
        }).toString()}`,
        feedDocumentSchema,
      ),
  };
}
```

```ts
  obj: object;
  msg: string | undefined;

export function recordingLogger(): IngestLogger & { entries: LogEntry[] } {
  const entries: LogEntry[] = [];
  const at =

    (obj: object, msg?: string): void => {
```

```ts
    felt: p.felt ?? null,
    cdi: p.cdi ?? null,
    mmi: p.mmi ?? null,
    tsunami: (p.tsunami ?? 0) !== 0,
    eventType: p.type ?? null,
    types: (p.types ?? "").split(",").filter((s) => s.length > 0),
    raw,
  };
}

export function parseFeedFeatures(doc: FeedDocument, log: IngestLogger): Event[] {
  const out: Event[] = [];
  for (const candidate of doc.features) {
    const parsed = featureSchema.safeParse(candidate);
    if (!parsed.success) {
      const featureId =
        typeof candidate === "object" && candidate !== null && "id" in candidate
          ? String((candidate as { id: unknown }).id)
          : undefined;
      log.warn({ featureId, issues: parsed.error.issues }, "skipping malformed feature");
      continue;
    }
    out.push(featureToEvent(parsed.data));
  }
  return out;
}

function num(props: Record<string, string>, key: string): number | null {
  const v = props[key];
  if (v === undefined || v === "") return null;
  const n = Number(v);
  return Number.isFinite(n) ? n : null;
}

export interface WeightedMechanism {
  mechanism: Mechanism;
  weight: number;
}

export function productToMechanism(
  eventId: string,
  kind: MechanismKind,
  product: Product,
): WeightedMechanism | null {
  if (product.status === "DELETE") return null;
  const p = product.properties;

  let strike1 = num(p, "nodal-plane-1-strike");
  let dip1 = num(p, "nodal-plane-1-dip");
  let rake1 = num(p, "nodal-plane-1-rake");
  let strike2 = num(p, "nodal-plane-2-strike");
  let dip2 = num(p, "nodal-plane-2-dip");
  let rake2 = num(p, "nodal-plane-2-rake");
  const tAzimuth = num(p, "t-axis-azimuth");
  const tPlunge = num(p, "t-axis-plunge");
  const pAzimuth = num(p, "p-axis-azimuth");
  const pPlunge = num(p, "p-axis-plunge");

  const noPlanes = [strike1, dip1, rake1, strike2, dip2, rake2].every((v) => v === null);
  const haveAxes = tAzimuth !== null && tPlunge !== null && pAzimuth !== null && pPlunge !== null;
  let planesReconstructed = false;
  if (noPlanes && haveAxes) {
    const [a, b] = planesFromAxes(
      { azimuth: tAzimuth, plunge: tPlunge },
      { azimuth: pAzimuth, plunge: pPlunge },
    );
    strike1 = a.strike;
    dip1 = a.dip;
    rake1 = a.rake;
    strike2 = b.strike;
    dip2 = b.dip;
    rake2 = b.rake;
    planesReconstructed = true;
  }

  const rawDc = num(p, "percent-double-couple");
  const percentDc = rawDc === null ? null : rawDc > 1 ? rawDc / 100 : rawDc;
  const magType = p["derived-magnitude-type"] ?? "";
  const mw = /^mw/i.test(magType) ? num(p, "derived-magnitude") : null;

  return {
    weight: product.preferredWeight ?? 0,
    mechanism: {
      eventId,
      kind,
      source: product.source,
      code: product.code,
      preferred: false,
      strike1,
      dip1,
      rake1,
      strike2,
      dip2,
      rake2,
      tAzimuth,
      tPlunge,
      pAzimuth,
      pPlunge,
      nAzimuth: num(p, "n-axis-azimuth"),
      nPlunge: num(p, "n-axis-plunge"),
      mrr: num(p, "tensor-mrr"),
      mtt: num(p, "tensor-mtt"),
      mpp: num(p, "tensor-mpp"),
      mrt: num(p, "tensor-mrt"),
      mrp: num(p, "tensor-mrp"),
      mtp: num(p, "tensor-mtp"),
      scalarMoment: num(p, "scalar-moment"),
      mw,
      depthKm: num(p, "derived-depth") ?? num(p, "depth"),
      percentDc,
      planesReconstructed,
      productUpdated: new Date(product.updateTime),
      raw: p,
    },
  };
}

/** Marks the heaviest product of each kind as preferred; ties go to the latest update. */
export function pickPreferred(items: WeightedMechanism[]): Mechanism[] {
  const best = new Map<MechanismKind, WeightedMechanism>();
  for (const item of items) {
    const cur = best.get(item.mechanism.kind);
    const later =
      (item.mechanism.productUpdated?.getTime() ?? 0) >
      (cur?.mechanism.productUpdated?.getTime() ?? 0);
    if (!cur || item.weight > cur.weight || (item.weight === cur.weight && later)) {
      best.set(item.mechanism.kind, item);
    }
  }
  return items.map((item) => ({
    ...item.mechanism,
    preferred: best.get(item.mechanism.kind) === item,
  }));
}

const PRODUCT_KINDS: { key: string; kind: MechanismKind }[] = [
  { key: "moment-tensor", kind: "mt" },
  { key: "focal-mechanism", kind: "fm" },
];

export interface ParsedProducts {
  mechanisms: Mechanism[];
  deleted: MechanismKey[];
}

export function parseProducts(detail: DetailDocument, log: IngestLogger): ParsedProducts {
  const products = detail.properties.products ?? {};
  const weighted: WeightedMechanism[] = [];
  const deleted: MechanismKey[] = [];
  for (const { key, kind } of PRODUCT_KINDS) {
    for (const candidate of products[key] ?? []) {
      const parsed = productSchema.safeParse(candidate);
      if (!parsed.success) {
        log.warn(
          { eventId: detail.id, kind, issues: parsed.error.issues },
          "skipping malformed product",
        );
        continue;
      }
      const result = productToMechanism(detail.id, kind, parsed.data);
      if (result) weighted.push(result);
      else
        deleted.push({
          eventId: detail.id,
          kind,
          source: parsed.data.source,
          code: parsed.data.code,
        });
    }
  }
  return { mechanisms: pickPreferred(weighted), deleted };
}
```

```ts
import { upsertEvents, listDetailCandidates, markDetailChecked } from "../repositories/events.js";
import { deleteMechanism, upsertMechanism } from "../repositories/mechanisms.js";
import { featureToEvent, parseProducts } from "../usgs/parse.js";
import type { IngestContext } from "./context.js";

export interface DetailsOptions {
  limit?: number;
  concurrency?: number;
}

export interface DetailsResult {
  candidates: number;
  fetched: number;
  failed: number;
  mechanismsUpserted: number;
}

async function processCandidate(
  ctx: IngestContext,
  id: string,
  result: DetailsResult,
): Promise<void> {
  try {
    const detail = await ctx.usgs.fetchDetail(id);
    const event = featureToEvent(detail);
    const { mechanisms, deleted } = parseProducts(detail, ctx.log);
    await ctx.db.transaction(async (tx) => {
      await upsertEvents(tx, [event]);
      for (const m of mechanisms) await upsertMechanism(tx, m);
      for (const key of deleted) await deleteMechanism(tx, key);
      await markDetailChecked(tx, id, event.updated);
    });
    result.fetched += 1;
    result.mechanismsUpserted += mechanisms.length;
  } catch (err) {
    result.failed += 1;
    ctx.log.warn({ err, eventId: id }, "detail fetch failed; will retry next run");
  }
}

/** Fetches detail documents for events that advertise a mechanism and stores their products. */
export async function syncDetails(
  ctx: IngestContext,
  opts: DetailsOptions = {},
): Promise<DetailsResult> {
  const limit = opts.limit ?? 50;
  const concurrency = opts.concurrency ?? 2;
  const candidates = await listDetailCandidates(ctx.db, limit);
  const result: DetailsResult = {
    candidates: candidates.length,
    fetched: 0,
    failed: 0,
    mechanismsUpserted: 0,
  };
  let next = 0;
  const workers = Array.from({ length: Math.min(concurrency, candidates.length) }, async () => {
    while (next < candidates.length) {
      const c = candidates[next];
      next += 1;
      if (c) await processCandidate(ctx, c.id, result);
    }
  });
  await Promise.all(workers);
  return result;
}
```

```ts
  day: { kind: "2.5_day", run: "feed_day" },

  events: UpsertResult;
  details: DetailsResult;

  err instanceof Error ? err.message : String(err);

  const runId = await startRun(ctx.db, run, { feed: kind }, ctx.now());
  try {
    const doc = await ctx.usgs.fetchFeed(kind);
    const parsed = parseFeedFeatures(doc, ctx.log);
    const events = await upsertEvents(ctx.db, parsed);
    const details = await syncDetails(ctx);
    await finishRun(
      ctx.db,
      runId,

        eventsInserted: events.inserted,
        eventsUpdated: events.updated,
        eventsSkipped: events.skipped,
        detailsFetched: details.fetched,

      },
      ctx.now(),
    );
    ctx.log.info({ runId, feed: kind, events, details }, "feed sync complete");
    return { runId, events, details };
  } catch (err) {
    await finishRun(ctx.db, runId, { ok: false, error: errorMessage(err) }, ctx.now());

  }
```

```ts
import { upsertEvents, type UpsertResult } from "../repositories/events.js";
import { finishRun, startRun } from "../repositories/ingest-runs.js";
import { parseFeedFeatures } from "../usgs/parse.js";
import type { IngestContext } from "./context.js";
import { syncDetails, type DetailsResult } from "./details.js";
import { errorMessage } from "./feed.js";

export interface BackfillOptions {
  days: number;
  windowDays?: number;
  minMagnitude?: number;
  pageLimit?: number;
}

export interface BackfillResult {
  runId: number;
  pages: number;
  events: UpsertResult;
  details: DetailsResult;
}

const DAY_MS = 86_400_000;

const add = (a: UpsertResult, b: UpsertResult): UpsertResult => ({
  inserted: a.inserted + b.inserted,
  updated: a.updated + b.updated,
  skipped: a.skipped + b.skipped,
});

/** Loads one time window page by page, restarting each page from the last event's time. */
async function loadWindow(
  ctx: IngestContext,
  start: Date,
  end: Date,
  minmagnitude: number,
  limit: number,
  totals: { pages: number; events: UpsertResult },
): Promise<void> {
  let cursor = start;
  for (;;) {
    const doc = await ctx.usgs.fetchQuery({ starttime: cursor, endtime: end, minmagnitude, limit });
    const events = parseFeedFeatures(doc, ctx.log);
    totals.pages += 1;
    totals.events = add(totals.events, await upsertEvents(ctx.db, events));
    if (events.length < limit) return;
    const last = events[events.length - 1];
    if (!last || last.time.getTime() <= cursor.getTime()) {
      throw new Error(`backfill cursor cannot advance past ${cursor.toISOString()}`);
    }
    cursor = last.time;
  }
}

export async function backfill(ctx: IngestContext, opts: BackfillOptions): Promise<BackfillResult> {
  const windowDays = opts.windowDays ?? 30;
  const minMagnitude = opts.minMagnitude ?? 2.5;
  const pageLimit = opts.pageLimit ?? 20_000;
  const end = ctx.now();
  const start = new Date(end.getTime() - opts.days * DAY_MS);
  const runId = await startRun(
    ctx.db,
    "backfill",
    {
      days: opts.days,
      windowDays,
      minMagnitude,
      pageLimit,
      start: start.toISOString(),
      end: end.toISOString(),
    },
    end,
  );
  try {
    const totals = { pages: 0, events: { inserted: 0, updated: 0, skipped: 0 } };
    for (let ws = start; ws < end; ws = new Date(ws.getTime() + windowDays * DAY_MS)) {
      const we = new Date(Math.min(ws.getTime() + windowDays * DAY_MS, end.getTime()));
      await loadWindow(ctx, ws, we, minMagnitude, pageLimit, totals);
    }
    const details: DetailsResult = { candidates: 0, fetched: 0, failed: 0, mechanismsUpserted: 0 };
    for (;;) {
      const r = await syncDetails(ctx, { limit: 200 });
      details.candidates += r.candidates;
      details.fetched += r.fetched;
      details.failed += r.failed;
      details.mechanismsUpserted += r.mechanismsUpserted;
      if (r.candidates === 0 || r.fetched === 0) break;
    }
    await finishRun(
      ctx.db,
      runId,
      {
        ok: true,
        eventsInserted: totals.events.inserted,
        eventsUpdated: totals.events.updated,
        eventsSkipped: totals.events.skipped,
        detailsFetched: details.fetched,
        mechanismsUpserted: details.mechanismsUpserted,
      },
      ctx.now(),
    );
    ctx.log.info(
      { runId, pages: totals.pages, events: totals.events, details },
      "backfill complete",
    );
    return { runId, pages: totals.pages, events: totals.events, details };
  } catch (err) {
    await finishRun(ctx.db, runId, { ok: false, error: errorMessage(err) }, ctx.now());
    throw err;
  }
}
```

```ts
import type { IngestContext } from "./context.js";
import { syncFeed, type FeedName, type FeedRunResult } from "./feed.js";

export interface SchedulerOptions {
  dayMs: number;
  weekMs: number;
  /** Injectable for tests; defaults to syncFeed. */
  run?: (ctx: IngestContext, name: FeedName) => Promise<FeedRunResult>;
}

export interface Scheduler {
  stop: () => Promise<void>;
}

/**
 * Runs the day feed now and every `dayMs`, the week feed every `weekMs`.
 * Ticks never overlap: if a run is in progress the tick is skipped.
 */
export function startScheduler(ctx: IngestContext, opts: SchedulerOptions): Scheduler {
  const run = opts.run ?? syncFeed;
  const timers = new Set<NodeJS.Timeout>();
  let inFlight: Promise<void> | null = null;
  let stopped = false;

  const trigger = (name: FeedName): void => {
    if (inFlight) {
      ctx.log.warn({ feed: name }, "skipping tick: previous run still in progress");
      return;
    }
    inFlight = run(ctx, name)
      .then(
        () => undefined,
        (err: unknown) => {
          ctx.log.error({ err, feed: name }, "scheduled ingest run failed");
        },
      )
      .finally(() => {
        inFlight = null;
      });
  };

  const schedule = (name: FeedName, ms: number): void => {
    if (stopped) return;
    const timer = setTimeout(() => {
      timers.delete(timer);
      trigger(name);
      schedule(name, ms);
    }, ms);
    timers.add(timer);
  };

  trigger("day");
  schedule("day", opts.dayMs);
  schedule("week", opts.weekMs);

  return {
    stop: async () => {
      stopped = true;
      for (const timer of timers) clearTimeout(timer);
      timers.clear();
      if (inFlight) await inFlight;
    },
  };
}
```

```ts
  options: {
    days: { type: "string" },
    "window-days": { type: "string" },
    limit: { type: "string" },
  },

const config = loadConfig();

const { db, close } = createDb(config.DATABASE_URL);

const int = (v: string | undefined, name: string): number | undefined => {
  if (v === undefined) return undefined;
  const n = Number(v);
  if (!Number.isInteger(n) || n < 1) throw new Error(`${name} must be a positive integer`);
  return n;
};

try {
  const [command, arg] = positionals;
  if (command === "feed" && (arg === "day" || arg === "week")) {
    const r = await syncFeed(ctx, arg);
    log.info(r, "feed done");
  } else if (command === "details") {
    const limit = int(values.limit, "--limit");
    const r = await syncDetails(ctx, limit === undefined ? {} : { limit });

    console.error(usage);
    process.exitCode = 2;
  }
} catch (err) {
  log.error({ err }, "ingest failed");
  process.exitCode = 1;
} finally {
  await close();
```
